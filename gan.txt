import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers,models,optimizers,datasets
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout, Activation, Flatten, Conv2D, MaxPooling2D
import numpy as np
import matplotlib.pyplot as plt
     

# load MNIST
(x_train, y_train), (x_test, y_test) = datasets.mnist.load_data()

# normalize and reshape
x_train = x_train / 255.0
x_train = x_train.reshape(-1, 784)

# generator
generator = Sequential([
    layers.Dense(128, activation='relu', input_shape=(100,)),
    layers.Dense(784, activation='sigmoid')
])

# discriminator
discriminator = Sequential([
    layers.Dense(128, activation='relu', input_shape=(784,)),
    layers.Dense(1, activation='sigmoid')
])

discriminator.compile(optimizer='adam', loss='binary_crossentropy')

# training
for epoch in range(100):

    # real images
    index = np.random.randint(0, x_train.shape[0], 32)
    real_images = x_train[index]

    # fake images
    noise = np.random.randn(32,100)
    fake_images = generator.predict(noise, verbose=0)

    # train discriminator
    discriminator.train_on_batch(real_images, np.ones((32,1)))
    discriminator.train_on_batch(fake_images, np.zeros((32,1)))

print("GAN Training Done")

# generate image
noise = np.random.randn(1,100)
generated_image = generator.predict(noise, verbose=0).reshape(28,28)

# show image
plt.imshow(generated_image, cmap='gray')
plt.title("Generated Image")
plt.show()
